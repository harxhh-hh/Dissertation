# Runtime dependencies.
#
# Each dependency is justified in README.md ("Dependencies and their
# justification"). Versions are pinned to a compatible-release range so that
# the environment can be reconstructed by the supervisor and moderator.
#
# Install with:  pip install -r requirements.txt

# Official Anthropic SDK. Provides the Messages API client, typed response
# objects (token usage, stop reasons), streaming, and automatic retry with
# exponential backoff on HTTP 429/5xx.
anthropic>=0.75,<1.0

# Reads key=value pairs from a local .env file into the process environment,
# so that the API key is never written into source code or committed.
python-dotenv>=1.0,<2.0

# Official Groq SDK. Provides an OpenAI-compatible Chat Completions client
# (`response_format={"type": "json_object"}` for structured output,
# `prompt_tokens`/`completion_tokens` usage fields) and automatic retry
# with exponential backoff on HTTP 429/5xx. Used when LLM_PROVIDER=groq.
# Only one of `anthropic`, `groq`, or the stdlib-only `ollama` backend is
# exercised per run; all remain available so the provider can be swapped
# by editing .env alone.
groq>=0.9,<1.0

# --- UI (optional: only needed to run `streamlit run ui/app.py`) ----------
# The dashboard that wraps run_experiment.py with a browser UI and live log
# streaming. Not required for the core CLI pipeline or the smoke test.
streamlit>=1.38,<2.0

# Renders a generated SRS Markdown document to a clean, self-contained HTML
# page for the UI's "view" feature (and standalone via ui/render_html.py).
markdown>=3.6,<4.0
